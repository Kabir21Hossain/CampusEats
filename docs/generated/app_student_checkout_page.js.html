"use client";

import React, { useEffect, useRef, useState, Suspense } from "react";
import Link from "next/link";
import { useRouter, useSearchParams } from "next/navigation";
import { useCart } from "@/context/cart-context";
import { useAuth } from "@/context/auth-context";
import { orderService } from "@/services/order-service";
import { CAMPUS_LOCATIONS } from "@/lib/constants";
import { Card, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { formatCurrency } from "@/lib/utils";
import {
  MapPin,
  CreditCard,
  ShieldCheck,
  Building,
  ArrowRight,
  Bike,
} from "lucide-react";
import confetti from "canvas-confetti";
import { toast } from "sonner";

/**
 * Function CheckoutContent.
 * @returns {*} The operation result.
 */
function CheckoutContent() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const noteParam = searchParams.get("note") || "";

  const { items, shop, subtotal, deliveryFee, platformFee, total, clearCart } = useCart();
  const { currentUser } = useAuth();

  const [deliveryHall, setDeliveryHall] = useState(CAMPUS_LOCATIONS[0]);
  const [roomNumber, setRoomNumber] = useState("");
  const [deliveryNote, setDeliveryNote] = useState(noteParam);
  const [isPlacing, setIsPlacing] = useState(false);
  const orderAttemptRef = useRef(null);

  useEffect(() => {
    if (currentUser?.hall && CAMPUS_LOCATIONS.includes(currentUser.hall)) {
      setDeliveryHall(currentUser.hall);
    }
    if (currentUser?.roomNumber) setRoomNumber(currentUser.roomNumber);
  }, [currentUser?.id, currentUser?.hall, currentUser?.roomNumber]);

  if (items.length === 0) {
    return (
      <div className="py-20 text-center space-y-4">
        <h2 className="text-xl font-bold">Your cart is empty</h2>
        <Link href="/student/home">
          <Button variant="primary">Browse Campus Food</Button>
        </Link>
      </div>
    );
  }

  /**
   * Arrow function handleInitiateOrder.
   * @param {*} e - Input value used by the operation.
   * @returns {*} The operation result.
   */
  const handleInitiateOrder = (e) => {
    e.preventDefault();
    executeOrderCreation();
  };

  /**
   * Arrow function executeOrderCreation.
   * @returns {Promise<*>} Resolves with the operation result.
   */
  const executeOrderCreation = async () => {
    setIsPlacing(true);
    const orderPayload = {
      shopId: shop?.id,
      deliveryAddress: `${deliveryHall}, ${roomNumber}`,
      deliveryNote,
      items: items.map((i) => ({
        menuItemId: i.item.id,
        name: i.item.name,
        quantity: i.quantity,
        priceAtOrder: i.item.priceBDT,
      })),
      subtotal,
      deliveryFee,
      total,
      paymentMethod: "bKash",
    };

    const requestFingerprint = JSON.stringify({
      shopId: orderPayload.shopId,
      items: orderPayload.items.map((item) => ({
        id: item.menuItemId,
        quantity: item.quantity,
      })),
      deliveryLocation: orderPayload.deliveryAddress.trim(),
      deliveryNotes: orderPayload.deliveryNote.trim(),
      paymentMethod: orderPayload.paymentMethod,
    });
    if (orderAttemptRef.current?.fingerprint !== requestFingerprint) {
      const idempotencyKey = window.crypto?.randomUUID?.();
      if (!idempotencyKey) {
        toast.error("Secure order submission is unavailable in this browser.");
        setIsPlacing(false);
        return;
      }
      orderAttemptRef.current = { fingerprint: requestFingerprint, idempotencyKey };
    }

    try {
      const res = await orderService.createOrder(
        orderPayload,
        orderAttemptRef.current.idempotencyKey
      );
      if (res.success) {
        orderAttemptRef.current = null;
        if (res.meta?.notificationWarning) toast.warning(res.meta.notificationWarning);
        // Trigger celebratory confetti
        try {
          confetti({
            particleCount: 80,
            spread: 70,
            origin: { y: 0.6 },
          });
        } catch {
          // ignore
        }

        toast.success(`Order #${res.data.id} placed successfully!`);
        clearCart();
        router.push(`/student/orders/${res.data.id}`);
      } else {
        toast.error(res.error?.message || "Failed to create order");
      }
    } catch (err) {
      toast.error("An unexpected error occurred");
    } finally {
      setIsPlacing(false);
    }
  };

  return (
    <div className="max-w-4xl mx-auto space-y-8 pb-16">
      <div>
        <h1 className="text-2xl font-bold text-zinc-900 dark:text-zinc-50">
          Confirm Delivery &amp; Checkout
        </h1>
        <p className="text-xs text-zinc-500 dark:text-zinc-400 mt-1">
          Review your dormitory address and select your payment method
        </p>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-12 gap-8 items-start">
        {/* Left Form: Delivery Address & Payment */}
        <form onSubmit={handleInitiateOrder} className="lg:col-span-7 space-y-6">
          {/* Dorm Delivery Details */}
          <Card>
            <CardContent className="p-6 space-y-4">
              <div className="flex items-center gap-2 pb-3 border-b border-zinc-100 dark:border-zinc-800">
                <MapPin className="h-4 w-4 text-emerald-600" />
                <h3 className="font-bold text-sm text-zinc-900 dark:text-zinc-100">
                  Dormitory Destination
                </h3>
              </div>

              <div className="space-y-1.5 text-left">
                <label className="text-xs font-semibold text-zinc-700 dark:text-zinc-300">
                  Select Residential Hall / Landmark
                </label>
                <select
                  value={deliveryHall}
                  onChange={(e) => setDeliveryHall(e.target.value)}
                  className="w-full rounded-xl border border-zinc-200 dark:border-zinc-800 bg-white/90 dark:bg-zinc-900/90 px-3.5 py-2.5 text-sm text-zinc-900 dark:text-zinc-100 focus:outline-none focus:border-emerald-600"
                >
                  {CAMPUS_LOCATIONS.map((loc) => (
                    <option key={loc} value={loc}>
                      {loc}
                    </option>
                  ))}
                </select>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <Input
                  label="Room & Floor Number"
                  placeholder="e.g. Room 304, 3rd Floor"
                  icon={Building}
                  value={roomNumber}
                  onChange={(e) => setRoomNumber(e.target.value)}
                  required
                />

              </div>

              <Input
                label="Delivery Instructions"
                placeholder="e.g. Leave with hall security if attending lecture..."
                value={deliveryNote}
                onChange={(e) => setDeliveryNote(e.target.value)}
              />
            </CardContent>
          </Card>

          {/* Payments are submitted and verified as two independent obligations. */}
          <Card>
            <CardContent className="p-6 space-y-4">
              <div className="flex items-center gap-2 pb-3 border-b border-zinc-100 dark:border-zinc-800">
                <CreditCard className="h-4 w-4 text-emerald-600" />
                <h3 className="font-bold text-sm text-zinc-900 dark:text-zinc-100">
                  bKash payment
                </h3>
              </div>

              <p className="text-xs text-zinc-600 dark:text-zinc-300">
                After the order is created, send the food amount to the shop and
                the delivery plus platform fees to CampusEats. Submit each
                transaction reference separately. References stay pending until
                an administrator verifies merchant records. Never enter your
                bKash PIN or OTP here.
              </p>
            </CardContent>
          </Card>

          <Button
            type="submit"
            variant="primary"
            size="lg"
            isLoading={isPlacing}
            className="w-full rounded-2xl py-3.5 text-base font-bold shadow-md"
          >
            <span>Confirm &amp; Place Order ({formatCurrency(total)})</span>
            <ArrowRight className="h-5 w-5 ml-1" />
          </Button>
        </form>

        {/* Right Summary */}
        <div className="lg:col-span-5 space-y-4">
          <Card>
            <CardContent className="p-6 space-y-4">
              <h3 className="font-bold text-sm text-zinc-900 dark:text-zinc-100 pb-2 border-b border-zinc-100 dark:border-zinc-800">
                Order Items ({items.length})
              </h3>

              <div className="space-y-3 max-h-60 overflow-y-auto divide-y divide-zinc-100 dark:divide-zinc-800/60 pr-1">
                {items.map(({ item, quantity }) => (
                  <div key={item.id} className="pt-2 flex justify-between items-center text-xs">
                    <div>
                      <p className="font-bold text-zinc-900 dark:text-zinc-100">
                        {quantity}x {item.name}
                      </p>
                      <p className="text-zinc-400">{formatCurrency(item.priceBDT)} each</p>
                    </div>
                    <span className="font-bold text-zinc-900 dark:text-zinc-100">
                      {formatCurrency(item.priceBDT * quantity)}
                    </span>
                  </div>
                ))}
              </div>

              <div className="pt-3 border-t border-zinc-100 dark:border-zinc-800 space-y-2 text-xs text-zinc-600 dark:text-zinc-300">
                <div className="flex justify-between">
                  <span>Subtotal</span>
                  <span className="font-semibold text-zinc-900 dark:text-zinc-100">
                    {formatCurrency(subtotal)}
                  </span>
                </div>
                <div className="flex justify-between">
                  <span className="flex items-center gap-1">
                    <Bike className="h-3.5 w-3.5 text-emerald-600" />
                    <span>Hall Delivery Fee</span>
                  </span>
                  <span className="font-semibold text-zinc-900 dark:text-zinc-100">
                    {formatCurrency(deliveryFee)}
                  </span>
                </div>
                <div className="flex justify-between">
                  <span>Platform Fee</span>
                  <span className="font-semibold text-zinc-900 dark:text-zinc-100">
                    {formatCurrency(platformFee)}
                  </span>
                </div>
                <div className="pt-2 border-t border-zinc-100 dark:border-zinc-800 flex justify-between text-base font-extrabold text-zinc-900 dark:text-zinc-50">
                  <span>Total Amount</span>
                  <span className="text-emerald-700 dark:text-emerald-400">
                    {formatCurrency(total)}
                  </span>
                </div>
              </div>

              <div className="p-3 rounded-xl bg-emerald-50/80 dark:bg-emerald-950/40 border border-emerald-200/60 dark:border-emerald-800/60 flex items-start gap-2 text-xs text-emerald-800 dark:text-emerald-300">
                <ShieldCheck className="h-4 w-4 shrink-0 mt-0.5" />
                <span className="leading-tight">
                  Prices and fees are recalculated by the server. Order preparation
                  cannot begin until both payment obligations are verified.
                </span>
              </div>
            </CardContent>
          </Card>
        </div>
      </div>

    </div>
  );
}

/**
 * Function StudentCheckoutPage.
 * @returns {*} The operation result.
 */
export default function StudentCheckoutPage() {
  return (
    <Suspense fallback={<div className="p-8 text-center text-xs">Loading checkout...</div>}>
      <CheckoutContent />
    </Suspense>
  );
}
